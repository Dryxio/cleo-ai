// CLEO5 custom script: Vanilla Mission Selector
// Sanny Builder 4
// mode: GTA SA (v1.0 - SBL)
//
// Launches missions declared in data\script\main.scm using
// LOAD_AND_LAUNCH_MISSION_INTERNAL instead of custom .cm missions.

{$CLEO .cs}
{$USE CLEO+}

const
    Menu_Key = KeyCode.F6
    Accept_Key = KeyCode.E
    Close_Key = KeyCode.Escape
    Mission_Count = 135
    Rows_Per_Page = 10
end

script_name {name} 'mis_sel'

wait {time} 2000
use_text_commands {state} true

int menuOpen = false
int currentMissionId = 11
longString selectedMissionName = ""

while true
    wait {time} 0

    if
        not is_player_playing $player1
    then
        menuOpen = false
        continue
    end

    if
        is_key_pressed {keyCode} Menu_Key
    then
        menuOpen = 1 - menuOpen
        while is_key_pressed {keyCode} Menu_Key
            wait {time} 0
        end
    end

    if and
        menuOpen == false
        test_cheat {input} "missions"
    then
        menuOpen = true
    end

    if
        menuOpen == true
    then
        DRAW_MENU(currentMissionId)

        if
            is_key_pressed {keyCode} KeyCode.Up
        then
            currentMissionId -= 1
            currentMissionId = WRAP_MISSION_ID(currentMissionId)
            while is_key_pressed {keyCode} KeyCode.Up
                wait {time} 0
            end
        end

        if
            is_key_pressed {keyCode} KeyCode.Down
        then
            currentMissionId += 1
            currentMissionId = WRAP_MISSION_ID(currentMissionId)
            while is_key_pressed {keyCode} KeyCode.Down
                wait {time} 0
            end
        end

        if
            is_key_pressed {keyCode} KeyCode.Left
        then
            currentMissionId -= Rows_Per_Page
            currentMissionId = WRAP_MISSION_ID(currentMissionId)
            while is_key_pressed {keyCode} KeyCode.Left
                wait {time} 0
            end
        end

        if
            is_key_pressed {keyCode} KeyCode.Right
        then
            currentMissionId += Rows_Per_Page
            currentMissionId = WRAP_MISSION_ID(currentMissionId)
            while is_key_pressed {keyCode} KeyCode.Right
                wait {time} 0
            end
        end

        if
            is_key_pressed {keyCode} Accept_Key
        then
            while is_key_pressed {keyCode} Accept_Key
                wait {time} 0
            end

            if
                CAN_LAUNCH_VANILLA_MISSION()
            then
                selectedMissionName = GET_MISSION_NAME(currentMissionId)
                menuOpen = false
                LAUNCH_VANILLA_MISSION(currentMissionId, selectedMissionName)
            else
                print_help_string {text} "~r~Cannot start a mission right now.~n~~w~Be on foot and out of other missions."
            end
        end

        if
            is_key_pressed {keyCode} Close_Key
        then
            menuOpen = false
            while is_key_pressed {keyCode} Close_Key
                wait {time} 0
            end
        end
    end
end

terminate_this_custom_script


function DRAW_MENU(currentMissionId: int)
    int page = currentMissionId
    int totalPages = Mission_Count
    int pageStart = currentMissionId
    int rowMissionId

    page /= Rows_Per_Page
    page += 1

    totalPages -= 1
    totalPages /= Rows_Per_Page
    totalPages += 1

    pageStart /= Rows_Per_Page
    pageStart *= Rows_Per_Page

    draw_rect {pos} 320.0 190.0 {size} 540.0 308.0 {rgb} 0x00 0x00 0x00 {alpha} 190
    draw_rect {pos} 320.0 58.0 {size} 540.0 34.0 {rgb} 0x1A 0x1A 0x1A {alpha} 220

    set_text_font {font} Font.Menu
    set_text_scale {width} 0.44 {height} 1.35
    set_text_centre {state} true
    set_text_colour {rgb} 0xFF 0xD7 0x55 {alpha} 255
    set_text_edge {size} 1 {rgb} 0 0 0 {alpha} 220
    set_text_centre_size {width} 640.0
    display_text_formatted {pos} 320.0 44.0 {format} "Vanilla Mission Selector"

    set_text_scale {width} 0.24 {height} 0.92
    set_text_colour {rgb} 0xB8 0xDC 0xFF {alpha} 255
    display_text_formatted {pos} 320.0 78.0 {format} "Up/Down: browse   Left/Right: page   E: launch   F6 or Esc: close"

    rowMissionId = pageStart
    DRAW_ROW(rowMissionId, 108.0, currentMissionId)
    rowMissionId += 1
    DRAW_ROW(rowMissionId, 130.0, currentMissionId)
    rowMissionId += 1
    DRAW_ROW(rowMissionId, 152.0, currentMissionId)
    rowMissionId += 1
    DRAW_ROW(rowMissionId, 174.0, currentMissionId)
    rowMissionId += 1
    DRAW_ROW(rowMissionId, 196.0, currentMissionId)
    rowMissionId += 1
    DRAW_ROW(rowMissionId, 218.0, currentMissionId)
    rowMissionId += 1
    DRAW_ROW(rowMissionId, 240.0, currentMissionId)
    rowMissionId += 1
    DRAW_ROW(rowMissionId, 262.0, currentMissionId)
    rowMissionId += 1
    DRAW_ROW(rowMissionId, 284.0, currentMissionId)
    rowMissionId += 1
    DRAW_ROW(rowMissionId, 306.0, currentMissionId)

    set_text_scale {width} 0.23 {height} 0.88
    set_text_colour {rgb} 0xD0 0xD0 0xD0 {alpha} 220
    display_text_formatted {pos} 320.0 342.0 {format} "Mission ID %d / %d    Page %d / %d" {args} currentMissionId Mission_Count page totalPages
end


function DRAW_ROW(missionId: int, rowY: float, currentMissionId: int)
    longString missionName = ""
    longString rowText = ""
    float rowHighlightY = rowY

    if
        missionId >= Mission_Count
    then
        return
    end

    missionName = GET_MISSION_NAME(missionId)
    rowHighlightY += 8.0
    string_format rowText = "%03d  %s" {args} missionId missionName

    set_text_font {font} Font.Menu
    set_text_scale {width} 0.28 {height} 0.98
    set_text_centre {state} false
    set_text_edge {size} 1 {rgb} 0 0 0 {alpha} 220

    if
        missionId == currentMissionId
    then
        draw_rect {pos} 320.0 rowHighlightY {size} 478.0 18.0 {rgb} 0x56 0x3C 0x08 {alpha} 170
        set_text_colour {rgb} 0xFF 0xE4 0x8A {alpha} 255
        display_text_formatted {pos} 92.0 rowY {format} "> %s" {args} rowText
    else
        set_text_colour {rgb} 0xE0 0xE0 0xE0 {alpha} 235
        display_text_formatted {pos} 92.0 rowY {format} "  %s" {args} rowText
    end
end


function WRAP_MISSION_ID(missionId: int): int
    int lastMissionId = Mission_Count
    lastMissionId -= 1

    if
        missionId < 0
    then
        return lastMissionId
    end

    if
        missionId >= Mission_Count
    then
        return 0
    end

    return missionId
end


function CAN_LAUNCH_VANILLA_MISSION(): logical
    if or
        is_on_mission
        not can_player_start_mission $player1
        is_char_in_any_car $scplayer
        is_char_getting_in_to_a_car $scplayer
    then
        return false
    end

    return true
end


function LAUNCH_VANILLA_MISSION(missionId: int, missionName: string)
    clear_help
    clear_prints
    set_on_mission {status} true
    print_help_formatted {text} "~g~Launching~s~ %s" {args} missionName
    load_and_launch_mission_internal {index} missionId
end
