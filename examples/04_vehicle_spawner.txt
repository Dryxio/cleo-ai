// CLEO5 example: Vehicle Spawner with Functions
// Sanny Builder 4
// mode: GTA SA (v1.0 - SBL)
//
// Type "vspawn" cheat to spawn a random tuned vehicle.

{$CLEO .cs}

script_name {name} 'vspawn'

wait {time} 2000

while true
    wait {time} 250

    if and
        is_player_playing $player1
        is_player_control_on $player1
        test_cheat "vspawn"
    then
        // Pick random vehicle model
        int modelId = pick_random_int {values} #INFERNUS #TURISMO #BULLET #SULTAN #ELEGY #JESTER #URANUS #BANSHEE

        int car = SPAWN_VEHICLE(modelId)

        if
            car <> -1
        then
            // Warp player into car
            warp_char_into_car $scplayer {vehicle} car

            // Random colors
            int c1 = generate_random_int_in_range {min} 0 {max} 126
            int c2 = generate_random_int_in_range {min} 0 {max} 126
            change_car_colour car {color1} c1 {color2} c2

            print_help_string {text} "~g~Vehicle spawned!"
        end
    end
end

terminate_this_script


function SPAWN_VEHICLE(modelId: int): int
    float x, y, z
    x, y, z = get_char_coordinates $scplayer

    // Get position ahead of player
    float heading = get_char_heading $scplayer
    x, y, z = get_offset_from_char_in_world_coords $scplayer {offset} 0.0 5.0 0.0

    // Remove player from any current vehicle
    if
        is_char_in_any_car $scplayer
    then
        warp_char_from_car_to_coord $scplayer {pos} x y z
    end

    request_model {modelId} modelId
    load_all_models_now

    int car = create_car {modelId} modelId {pos} x y z
    mark_model_as_no_longer_needed {modelId} modelId

    set_car_heading car {heading} heading

    // Improve vehicle
    improve_car_by_cheating car {state} true

    return car
end
